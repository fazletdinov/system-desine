package main

import (
	"fmt"
	"runtime"
	"runtime/metrics"
	"time"
)

type node struct{ refs [8]*node }

var sink []byte

func main() {
	live := make([]*node, (128<<20)/64)
	for i := range live {
		n := &node{}
		if i > 0 {
			n.refs[0] = live[i-1]
		}
		live[i] = n
	}
	stop := make(chan struct{})
	for i := 0; i < runtime.GOMAXPROCS(0); i++ {
		go func() {
			var s []byte
			for {
				select {
				case <-stop:
					return
				default:
				}
				s = make([]byte, 1024)
				s[0] = 1
			}
			_ = s
		}()
	}
	time.Sleep(3 * time.Second)
	close(stop)
	s := []metrics.Sample{{Name: "/gc/limiter/last-enabled:gc-cycle"}, {Name: "/gc/cycles/total:gc-cycles"}, {Name: "/gc/gomemlimit:bytes"}, {Name: "/cpu/classes/gc/total:cpu-seconds"}, {Name: "/cpu/classes/total:cpu-seconds"}}
	metrics.Read(s)
	fmt.Println("last-enabled", s[0].Value.Uint64(), "cycles", s[1].Value.Uint64(), "limit", s[2].Value.Uint64(), "gc share", s[3].Value.Float64()/s[4].Value.Float64())
	runtime.KeepAlive(live)
}
