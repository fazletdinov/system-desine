FROM messages m
JOIN chats c ON c.id = m.chat_id
JOIN users u ON u.id = m.author_id
ORDER BY m.id;

CREATE TABLE support_duty (
    client_id bigint NOT NULL,
    topic     text   NOT NULL,
    agent     text   NOT NULL,
    PRIMARY KEY (client_id, topic),
    UNIQUE (client_id, agent)
);

FROM client_agents ca
